(function(Scratch) {
    'use strict';

    let pyodide = null;
    let extensionInstance = null;

    class PythonRunnerExtension {
        getInfo() {
            return {
                id: 'pythonrunner',
                name: 'Python Communicator',
                color1: '#337a7a',
                menuIconURI: 'data:image/svg+xml;base64,PHN2ZyB4bWxucz0iaHR0cDovL3d3dy53My5vcmcvMjAwMC9zdmcgd2lkdGg9IjI0IiBoZWlnaHQ9IjI0IiB2aWV3Qm94PSIwIDAgMjQgMjQiPjxwYXRoIGZpbGw9IiNGRkZGRkYiIGQ9Ik0wIDBoMjR2MjRIMHoiLz48cGF0aCBmaWxsPSIjRkZGRkZGIiBkPSJNMTEgMTdIMTRWMTRIMTFWMTd6bTYgLTJIMTBBMCAwIDAgMCAxMCAwSDE3QTAgMCAwIDAgMTcgMTV6bS00LTZoLTNWMTFoM1Y5em0zIDJIMTBBMCAwIDAgMCAxMCA3SDE2QTAgMCAwIDAgMTYgOXoiLz48L3N2Zz4=',
                blocks: [
                    {
                        opcode: 'runPythonCode',
                        blockType: Scratch.BlockType.REPORTER,
                        text: 'Ask Python to run script: [CODE]',
                        arguments: {
                            CODE: {
                                type: Scratch.ArgumentType.STRING,
                                defaultValue: 'result = 5 + 5\nprint(result)'
                            }
                        }
                    },
                    {
                        opcode: 'getPyConsoleOutput',
                        blockType: Scratch.BlockType.REPORTER,
                        text: 'Hear the last thing Python said',
                    },
                    {
                        opcode: 'setPythonVariable',
                        blockType: Scratch.BlockType.COMMAND,
                        text: 'Tell Python variable [NAME] is now [VALUE]',
                        arguments: {
                            NAME: { type: Scratch.ArgumentType.STRING, defaultValue: 'my_var' },
                            VALUE: { type: Scratch.ArgumentType.STRING, defaultValue: '0' }
                        }
                    },
                    {
                        opcode: 'getPythonVariable',
                        blockType: Scratch.BlockType.REPORTER,
                        text: 'Listen for Python variable [NAME]',
                        arguments: {
                            NAME: { type: Scratch.ArgumentType.STRING, defaultValue: 'my_var' }
                        }
                    },
                    {
                        opcode: 'installPackage',
                        blockType: Scratch.BlockType.COMMAND,
                        text: 'Add the knowledge package: [PACKAGE]',
                        arguments: {
                            PACKAGE: { type: Scratch.ArgumentType.STRING, defaultValue: 'numpy' }
                        }
                    },
                    {
                        opcode: 'runPythonFunction',
                        blockType: Scratch.BlockType.REPORTER,
                        text: 'Ask Python function [NAME] to use arguments: [ARGS]',
                        arguments: {
                            NAME: { type: Scratch.ArgumentType.STRING, defaultValue: 'add_numbers' },
                            ARGS: { type: Scratch.ArgumentType.STRING, defaultValue: '[]' }
                        }
                    },
                    {
                        opcode: 'clearPyConsoleOutput',
                        blockType: Scratch.BlockType.COMMAND,
                        text: 'Clear Python\'s memory of past speech',
                    },
                ]
            };
        }

        _lastOutput = ""; 

        async runPythonCode(args) {
            const code = args.CODE;
            if (pyodide === null) await initializePyodide();
            if (pyodide === null) return "Error: Pyodide not loaded.";

            try {
                this._lastOutput = ""; 
                let result = await pyodide.runPythonAsync(code, {
                    stdout: line => { this._lastOutput += line + '\n'; },
                    stderr: line => { this._lastOutput += "ERROR: " + line + '\n'; }
                });

                return String(result !== undefined ? result : "Code executed (no return value)");
            } catch (error) {
                console.error("Python runtime error:", error);
                this._lastOutput += `ERROR: ${error.message}\n`;
                return `Error: ${error.message.substring(0, 50)}...`;
            }
        }

        getPyConsoleOutput() {
            return this._lastOutput;
        }

        clearPyConsoleOutput() {
            this._lastOutput = "";
        }

        async setPythonVariable(args) {
            if (pyodide === null) await initializePyodide();
            if (pyodide === null) return; 

            const varName = args.NAME;
            const varValue = args.VALUE;

            try {
                await pyodide.runPythonAsync(`
                    import json
                    ${varName} = json.loads('''${JSON.stringify(varValue)}''')
                `);
            } catch (e) {
                console.error("Error setting Python variable:", e);
                this._lastOutput += `ERROR setting ${varName}: ${e.message}\n`;
            }
        }

        getPythonVariable(args) {
            if (pyodide === null) return "Python not loaded.";

            const varName = args.NAME;
            try {
                const value = pyodide.globals.get(varName);
                if (value === undefined) return `Variable '${varName}' not found`;
                return String(value);
            } catch (e) {
                console.error("Error getting Python variable:", e);
                return `Error getting ${varName}`;
            }
        }

        async installPackage(args) {
            if (pyodide === null) await initializePyodide();
            if (pyodide === null) return; 

            const packageName = args.PACKAGE;
            try {
                await pyodide.loadPackage(packageName);
                this._lastOutput += `Package '${packageName}' installed.\n`;
            } catch (e) {
                console.error("Error installing package:", e);
                this._lastOutput += `ERROR installing package ${packageName}: ${e.message}\n`;
            }
        }

        async runPythonFunction(args) {
            if (pyodide === null) await initializePyodide();
            if (pyodide === null) return "Error: Pyodide not loaded.";

            const funcName = args.NAME;
            const funcArgsJson = args.ARGS;

            try {
                const pyFunction = pyodide.globals.get(funcName);
                if (pyFunction === undefined) return `Function '${funcName}' not found`;

                const jsArgs = JSON.parse(funcArgsJson);

                const result = pyFunction(...jsArgs);
                return String(result);

            } catch (e) {
                console.error("Error running Python function:", e);
                this._lastOutput += `ERROR running function ${funcName}: ${e.message}\n`;
                return `Error: ${e.message.substring(0, 50)}...`;
            }
        }
    }

    async function initializePyodide() {
        if (pyodide !== null) return; 
        
        const script = document.createElement('script');
        script.src = "cdn.jsdelivr.net"; 
        document.head.appendChild(script);

        await new Promise(resolve => {
            script.onload = resolve;
            if (window.loadPyodide) resolve(); 
        });

        try {
            pyodide = await window.loadPyodide({
                indexURL: "cdn.jsdelivr.net",
            });
        } catch (error) {
            if (extensionInstance) {
                extensionInstance._lastOutput += `FATAL ERROR: Could not load Pyodide. Check internet connection. ${error.message}\n`;
            }
        }
    }
    
    Scratch.extensions.register(new PythonRunnerExtension());
})(Scratch);
